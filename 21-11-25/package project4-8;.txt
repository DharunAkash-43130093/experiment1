package project4;

public class OverloadDemo {
	public static void main(String args[]){
		basicc b = new basicc();
		b.add(1,2);
		b.add(1,2.2);
		b.add(3,5,6);
	}
	}
	//OVERLOADING
	class basicc{
		void add(int a,int b){
			System.out.println(a+b);
		}
		void add(int a,double b){
			System.out.println(a+b);
		}
		void add(int a,int b,int c){
			System.out.print(a+b+c);
		}
	}

class SBI {
    private String name;
    private String email;
    private long balance;

    public void setName(String name) {
        this.name = name;
    }

    public String getName() {
        return name;
    }

    public void setEmail(String email) {
        this.email = email;
    }

    public String getEmail() {
        return email;
    }

    public void setBalance(long balance) {
        this.balance = balance;
    }

    public long getBalance() {
        return balance;
    }
}
class PhonePe extends SBI {
    void checkDetails() {
        System.out.println(getName());
        System.out.println(getEmail());
        System.out.println(getBalance());
    }
}

public class Abstract {
	public static void main(String[] args) {
        HDFC h = new HDFC();
        h.agriLoan();
        h.eduLoan();

        DhanaLaxmi d = new DhanaLaxmi();
        d.agriLoan();
        d.eduLoan();
    }
}
abstract class RBI {
    private final void RBIBanking() {
        System.out.println("Secret..");
    }

    abstract void agriLoan();
    abstract void eduLoan();
}
class HDFC extends RBI {
    @Override
    void agriLoan() {
        System.out.println("HDFC Agri Loan: 12%");
    }
    @Override
    void eduLoan() {
        System.out.println("HDFC Education Loan: 10%");
    }
}
class DhanaLaxmi extends RBI {
    @Override
    void agriLoan() {
        System.out.println("DhanaLaxmi Agri Loan: 11%");
    }
    @Override
    void eduLoan() {
        System.out.println("DhanaLaxmi Education Loan: 9%");
    }
}
